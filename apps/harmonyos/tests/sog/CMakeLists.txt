cmake_minimum_required(VERSION 3.16)
project(SogHost C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CORE "${CMAKE_CURRENT_LIST_DIR}/../../entry/src/main/cpp")
add_subdirectory(${CORE}/third_party/libwebp webp EXCLUDE_FROM_ALL)
add_executable(sog-test ${CORE}/sog.cpp ${CORE}/splat.cpp test.cpp)
target_include_directories(sog-test PRIVATE ${CORE} ${CORE}/third_party ${CORE}/third_party/libwebp/src)
target_compile_options(sog-test PRIVATE -fsanitize=address,undefined -g)
target_link_options(sog-test PRIVATE -fsanitize=address,undefined)
target_link_libraries(sog-test webpdecoder z)
